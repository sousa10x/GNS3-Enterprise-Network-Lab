! ==========================================
! CONFIGURACAO INICIAL - SWITCH ACCESS 1
! ==========================================
enable
configure terminal

! 1. Criacao apenas das VLANs necessarias neste switch
vlan 10
 name Diretoria/Corp/TI
vlan 20
 name Comercial/Vendas
vlan 99
 name GESTAO_ZABBIX
vlan 321
 name NATIVA_MORTA
exit

! 2. Configuracao do IP de Gestao e Gateway (Para o Zabbix o encontrar)
interface vlan 99
 ip address 172.16.99.11 255.255.255.0
 no shutdown
exit
ip default-gateway 172.16.99.1

! 3. Porta Trunk (Uplink para o Core) - Com Seguranca (Sem VLAN 1)
interface Ethernet 0/0
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 321
 switchport trunk allowed vlan 10,20,99,321
 no shutdown
exit

! 4. Portas dos Utilizadores
! Computador da VLAN 20 (Interface 0/1)
interface Ethernet 0/1
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 no shutdown
exit

! Computador da VLAN 10 (Interface 0/2)
interface Ethernet 0/2
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 no shutdown
exit

! 5. Ativacao do SNMP para o Zabbix (Somente Leitura / RO)
! "public" e a senha/community, pode mudar para outra se preferir
snmp-server community lab RO
snmp-server location Rack-Andar-1
snmp-server contact Julio-Sousa
end
do write memory
