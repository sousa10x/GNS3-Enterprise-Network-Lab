! ==========================================
! CONFIGURACAO INICIAL - SWITCH CORE
! ==========================================

enable
configure terminal

! 1. Ativar Roteamento (Transforma o Core no "Roteador" da rede interna)
ip routing

! 2. Criacao das VLANs (Departamentos + Servidores + Seguranca)
vlan 10
 name Diretoria/Corp/TI
vlan 20
 name Comercial/Vendas
vlan 30
 name Recursos_Humanos
vlan 40
 name VISITANTES
vlan 99
 name SERVIDORES
vlan 321
 name NATIVA_MORTA
exit

! 3. Criacao dos Gateways (Avatares do Switch em cada rede)
interface vlan 10
 ip address 172.16.10.1 255.255.255.0
 no shutdown
interface vlan 20
 ip address 172.16.20.1 255.255.255.0
 no shutdown
interface vlan 30
 ip address 172.16.30.1 255.255.255.0
 no shutdown
interface vlan 40
 ip address 172.16.40.1 255.255.255.0
 no shutdown
interface vlan 99
 ip address 172.16.99.1 255.255.255.0
 no shutdown
exit

! 4. Portas Trunk (Para os Switches Access 1 e 2) - Com seguranca
interface range Ethernet 0/1 - 2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 321
 no shutdown
exit

! 5. Portas de Acesso (Onde vai ligar o AD, Zabbix, Radius, etc.)
interface range Ethernet 0/3 , Ethernet 1/0 - 2
 switchport mode access
 switchport access vlan 99
 spanning-tree portfast
 no shutdown
exit

! 6. Porta Roteada para o Firewall (pfSense)
interface Ethernet 0/0
 no switchport
 ip address 10.0.2.1 255.255.255.0
 no shutdown
exit
ip route 0.0.0.0 0.0.0.0 10.0.2.2

! 7. SNMP para o Zabbix monitorizar este switch
snmp-server community public RO
snmp-server location Rack-Core
snmp-server contact Julio-Sousa
end
do write memory
