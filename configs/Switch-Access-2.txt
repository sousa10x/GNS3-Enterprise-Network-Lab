! ==========================================
! CONFIGURACAO INICIAL - SWITCH ACCESS 2
! ==========================================
enable
configure terminal

! 1. Criacao apenas das VLANs necessarias neste switch
vlan 30
 name Recursos_Humanos
vlan 40
 name VISITANTES
vlan 99
 name GESTAO_ZABBIX
vlan 321
 name NATIVA_MORTA
exit

! 2. Configuracao do IP de Gestao e Gateway (Para o Zabbix o encontrar)
! (Utilizei o IP .12, assumindo que o Access 1 ficou com o .11)
interface vlan 99
 ip address 172.16.99.12 255.255.255.0
 no shutdown
exit
ip default-gateway 172.16.99.1

! 3. Porta Trunk (Uplink para o Core) - Com Seguranca
! Assumindo que a porta 0/0 é a que liga ao Switch Core
interface Ethernet 0/0
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 30,40,99,321
 no shutdown
exit

! 4. Portas dos Utilizadores
! Computador da VLAN 30 (Interface 0/1, conforme pediu)
interface Ethernet 0/1
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
 no shutdown
exit

! Computador da VLAN 40 (Interface 0/2, conforme pediu)
interface Ethernet 0/2
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
 no shutdown
exit

! 5. Ativacao do SNMP para o Zabbix
snmp-server community public RO
snmp-server location Rack-Access-2
snmp-server contact Julio-Sousa
end
do write memory
